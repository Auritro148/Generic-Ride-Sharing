#schema of user table

create table public.users (
  user_id uuid not null,
  first_name character varying(100) null,
  last_name character varying(100) null,
  phone character varying(20) null,
  email character varying(255) not null,
  hash_key character varying(255) not null,
  user_type character varying(50) null,
  is_verified boolean null default false,
  reg_date timestamp with time zone null default CURRENT_TIMESTAMP,
  updated_at timestamp with time zone null default CURRENT_TIMESTAMP,
  constraint users_pkey primary key (email),
  constraint users_user_id_key unique (user_id)
) TABLESPACE pg_default;



#schema of the driver table

create table public.drivers (
  driver_id uuid not null,
  active_vehicle_id bigint null,
  is_online boolean not null default false,
  constraint drivers_pkey primary key (driver_id),
  constraint fk_driver_active_vehicle foreign KEY (active_vehicle_id) references vehicles (vehicle_id)
) TABLESPACE pg_default;


#schema of driver_documents table 
create table public.driver_documents (
  document_id bigserial not null,
  driver_id uuid not null,
  vehicle_id bigint null,
  document_type character varying(50) not null,
  document_number character varying(100) null,
  file_url text not null,
  verified_status character varying(20) not null default 'PENDING'::character varying,
  uploaded_at timestamp without time zone not null default CURRENT_TIMESTAMP,
  created_at timestamp without time zone not null default CURRENT_TIMESTAMP,
  verified_at timestamp without time zone null,
  constraint driver_documents_pkey primary key (document_id),
  constraint fk_document_driver foreign KEY (driver_id) references drivers (driver_id),
  constraint fk_document_vehicle foreign KEY (vehicle_id) references vehicles (vehicle_id),
  constraint chk_document_status check (
    (
      (verified_status)::text = any (
        (
          array[
            'PENDING'::character varying,
            'VERIFIED'::character varying,
            'REJECTED'::character varying
          ]
        )::text[]
      )
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_driver_document_driver on public.driver_documents using btree (driver_id) TABLESPACE pg_default;



#schema of ride_requests
create table public.ride_requests (
  req_id bigint generated by default as identity not null,
  passenger_id uuid not null,
  timestamp timestamp without time zone not null default CURRENT_TIMESTAMP,
  seats integer not null default 1,
  vehicle_type_id bigint not null,
  est_fare numeric(10, 2) null,
  surge_multiplier numeric(5, 2) not null default 1.00,
  constraint ride_requests_pkey primary key (req_id),
  constraint fk_request_passenger foreign KEY (passenger_id) references passengers (passenger_id),
  constraint fk_request_vehicle_type foreign KEY (vehicle_type_id) references vehicle_types (type_id),
  constraint chk_est_fare check (
    (
      (est_fare is null)
      or (est_fare >= (0)::numeric)
    )
  ),
  constraint chk_request_seats check ((seats > 0)),
  constraint chk_surge_multiplier check ((surge_multiplier >= 1.00))
) TABLESPACE pg_default;

create index IF not exists idx_ride_request_passenger on public.ride_requests using btree (passenger_id) TABLESPACE pg_default;

create index IF not exists idx_ride_request_vehicle_type on public.ride_requests using btree (vehicle_type_id) TABLESPACE pg_default;


#schema of ride_request_location
create table public.ride_request_location (
  req_id bigint not null,
  pickup_lat double precision not null,
  pickup_long double precision not null,
  drop_lat double precision not null,
  drop_long double precision not null,
  time timestamp without time zone not null default CURRENT_TIMESTAMP,
  pick_addr text null,
  drop_addr text null,
  pick_name text null,
  drop_name text null,
  constraint ride_request_location_pkey primary key (req_id),
  constraint fk_ride_request foreign KEY (req_id) references ride_requests (req_id)
) TABLESPACE pg_default;

#schema of trip table
create table public.trips (
  trip_id bigserial not null,
  req_id bigint not null,
  driver_id uuid not null,
  vehicle_id bigint not null,
  pickup_addr text not null,
  dropoff_addr text not null,
  status character varying(30) not null,
  time_stamp timestamp without time zone not null default CURRENT_TIMESTAMP,
  pickup_time timestamp without time zone null,
  dropoff_time timestamp without time zone null,
  fare numeric(10, 2) null,
  constraint trips_pkey primary key (trip_id),
  constraint trips_req_id_key unique (req_id),
  constraint fk_trip_driver foreign KEY (driver_id) references drivers (driver_id),
  constraint fk_trip_vehicle foreign KEY (vehicle_id) references vehicles (vehicle_id),
  constraint chk_trip_fare check (
    (
      (fare is null)
      or (fare >= (0)::numeric)
    )
  ),
  constraint chk_trip_status check (
    (
      (status)::text = any (
        (
          array[
            'REQUESTED'::character varying,
            'CONFIRMED'::character varying,
            'ONGOING'::character varying,
            'COMPLETED'::character varying,
            'CANCELLED'::character varying
          ]
        )::text[]
      )
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_trip_vehicle on public.trips using btree (vehicle_id) TABLESPACE pg_default;

create index IF not exists idx_trip_driver on public.trips using btree (driver_id) TABLESPACE pg_default;
